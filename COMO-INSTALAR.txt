name: Gojo - Gerar cobrinha

on:
  workflow_dispatch:
  schedule:
    - cron: "53 5 * * *"
  push:
    branches: [main]
    paths:
      - "README.md"
      - ".github/workflows/snake-game.yml"

permissions:
  contents: write

jobs:
  gerar-cobrinha:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Gerar SVG animado com minhas contribuicoes
        uses: Platane/snk/svg-only@v3
        with:
          github_user_name: BernardoCipolli
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark

      - name: Publicar na branch output
        uses: crazy-max/ghaction-github-pages@v5
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
